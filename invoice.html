-- Tabel Invoice Subkontraktor
CREATE TABLE IF NOT EXISTS subcon_invoices (
    id SERIAL PRIMARY KEY,
    invoice_no VARCHAR(30) NOT NULL UNIQUE,
    contract_id INT NOT NULL REFERENCES subcon_contracts(id) ON DELETE CASCADE,
    progress_id INT REFERENCES subcon_progress(id),
    invoice_date DATE NOT NULL,
    due_date DATE,
    total_amount NUMERIC(15,2) NOT NULL DEFAULT 0,
    paid_amount NUMERIC(15,2) NOT NULL DEFAULT 0,
    status VARCHAR(20) NOT NULL DEFAULT 'Draft',
    reference_no VARCHAR(100),
    notes TEXT,
    created_by INT REFERENCES m_user_system(id),
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP,
    approved_by INT REFERENCES m_user_system(id),
    approved_at TIMESTAMP,
    cancelled_by INT REFERENCES m_user_system(id),
    cancelled_at TIMESTAMP,
    cancel_reason TEXT,
    CONSTRAINT chk_subcon_inv_status CHECK (status IN ('Draft', 'Diajukan', 'Disetujui', 'Dibatalkan')),
    CONSTRAINT chk_subcon_inv_amount CHECK (total_amount >= 0 AND paid_amount >= 0)
);

CREATE INDEX idx_subcon_invoices_contract ON subcon_invoices(contract_id);
CREATE INDEX idx_subcon_invoices_progress ON subcon_invoices(progress_id);
CREATE INDEX idx_subcon_invoices_status ON subcon_invoices(status);

COMMENT ON TABLE subcon_invoices IS 'Invoice subkontraktor (step 1: terima invoice dari subkon)';

-- Tabel Pembayaran Invoice Subkontraktor
CREATE TABLE IF NOT EXISTS subcon_invoice_payments (
    id SERIAL PRIMARY KEY,
    payment_no VARCHAR(30) NOT NULL UNIQUE,
    invoice_id INT NOT NULL REFERENCES subcon_invoices(id) ON DELETE CASCADE,
    contract_id INT NOT NULL REFERENCES subcon_contracts(id) ON DELETE CASCADE,
    payment_date DATE NOT NULL,
    account_id INT NOT NULL REFERENCES chart_of_accounts(id),
    payment_method VARCHAR(20) NOT NULL DEFAULT 'BANK_TRANSFER',
    amount NUMERIC(15,2) NOT NULL DEFAULT 0,
    reference_no VARCHAR(100),
    notes TEXT,
    status VARCHAR(20) NOT NULL DEFAULT 'Posted',
    created_by INT REFERENCES m_user_system(id),
    created_at TIMESTAMP DEFAULT NOW(),
    cancelled_by INT REFERENCES m_user_system(id),
    cancelled_at TIMESTAMP,
    cancel_reason TEXT,
    CONSTRAINT chk_subcon_inv_pay_status CHECK (status IN ('Posted', 'Dibatalkan')),
    CONSTRAINT chk_subcon_inv_pay_amount CHECK (amount > 0)
);

CREATE INDEX idx_subcon_inv_payments_invoice ON subcon_invoice_payments(invoice_id);
CREATE INDEX idx_subcon_inv_payments_contract ON subcon_invoice_payments(contract_id);

COMMENT ON TABLE subcon_invoice_payments IS 'Pembayaran invoice subkontraktor (step 2: release payment)';

function sc_inv_send($success, $data = [], $message = '', $extra = []) {
    $resp = array_merge(['success' => $success, 'message' => $message], $extra);
    if ($success) {
        $resp['data'] = $data;
        if (!isset($resp['totalCount'])) $resp['totalCount'] = is_array($data) ? count($data) : 0;
    }
    if (ob_get_length()) { ob_clean(); }
    echo json_encode($resp, JSON_INVALID_UTF8_SUBSTITUTE);
    exit;
}

                $db->commit();
                sc_inv_send(true, [], 'Invoice dibatalkan.');
            } catch (Throwable $e) {
                if ($db->inTransaction()) $db->rollBack();
                throw $e;
            }
            break;

        /* ------------------------------------------------------------------
         * STEP 2: LIST ACCOUNTS (untuk combo pembayaran)
         * ------------------------------------------------------------------ */
        case 'list_accounts':
            $rows = $db->query("SELECT id, account_code, account_name FROM chart_of_accounts
                WHERE account_type::text IN ('cash','bank') AND is_active
                ORDER BY account_code")->fetchAll(PDO::FETCH_ASSOC);
            foreach ($rows as &$r) {
                $r['id'] = (int)$r['id'];
                $r['label'] = $r['account_code'] . ' - ' . $r['account_name'];

            sc_inv_send(true, $rows);

         * STEP 2: SAVE PAYMENT (Release Payment)
         * ------------------------------------------------------------------ */
        case 'save_payment':
            $invoiceId = (int)($_POST['invoice_id'] ?? 0);
            $contractId = (int)($_POST['contract_id'] ?? 0);
            $paymentDate = (string)($_POST['payment_date'] ?? '');
            $accountId = (int)($_POST['account_id'] ?? 0);
            $paymentMethod = trim((string)($_POST['payment_method'] ?? 'BANK_TRANSFER'));
            $amount = sc_inv_f($_POST['amount'] ?? 0);

            if ($accountId <= 0) throw new ScInvError('Akun kas/bank wajib dipilih.');
            if ($amount <= 0) throw new ScInvError('Jumlah pembayaran harus lebih dari 0.');
            if ($paymentDate === '') throw new ScInvError('Tanggal pembayaran wajib diisi.');
            
            $db->beginTransaction();
            try {
                $stmt = $db->prepare("SELECT * FROM subcon_invoices WHERE id = :id FOR UPDATE");
                $stmt->execute([':id' => $invoiceId]);
                $inv = $stmt->fetch(PDO::FETCH_ASSOC);
                if (!$inv) throw new ScInvError('Invoice tidak ditemukan.');
                if ($inv['status'] !== 'Disetujui') throw new ScInvError('Hanya invoice Disetujui yang bisa dibayar.');
                if ((int)$inv['contract_id'] !== $contractId) throw new ScInvError('Invoice tidak sesuai kontrak.');
                
                $stmt = $db->prepare("SELECT COALESCE(SUM(amount), 0) FROM subcon_invoice_payments WHERE invoice_id = :id AND status = 'Posted'");
                $stmt->execute([':id' => $invoiceId]);
                $paid = (float)$stmt->fetchColumn();
                $remaining = round((float)$inv['total_amount'] - $paid, 2);
                
                if ($amount > $remaining + SC_INV_EPS) {
                    throw new ScInvError('Jumlah pembayaran melebihi sisa hutang invoice (sisa: ' . number_format($remaining, 2) . ').');
                }
                
                if (!function_exists('CreateRefNo')) throw new Exception('CreateRefNo() tidak tersedia.');
                $payNo = CreateRefNo('subcon_invoice_payments', 'payment_no', 'PAY-SC');
                
                $stmt = $db->prepare("INSERT INTO subcon_invoice_payments (payment_no, invoice_id, contract_id, payment_date,
                    account_id, payment_method, amount, reference_no, notes, status, created_by)
                    VALUES (:no, :iid, :cid, CAST(:pdate AS date), :aid, :method, :amt, :ref, :notes, 'Posted', :uid)
                    RETURNING id");
                $stmt->execute([':no' => $payNo, ':iid' => $invoiceId, ':cid' => $contractId,
                    ':pdate' => $paymentDate, ':aid' => $accountId, ':method' => $paymentMethod,
                    ':amt' => $amount, ':ref' => $referenceNo, ':notes' => $notes, ':uid' => sc_inv_uid()]);
                $payId = (int)$stmt->fetchColumn();
                
                $newPaid = round($paid + $amount, 2);
                $db->prepare("UPDATE subcon_invoices SET paid_amount = :paid WHERE id = :id")
                    ->execute([':paid' => $newPaid, ':id' => $invoiceId]);
                
                $desc = "Pembayaran invoice subkon {$inv['invoice_no']}";
                sc_inv_post($db, 'SUBCON_INVOICE_PAY', SC_INV_PAY_REF, $payId, $paymentDate, $amount, $desc, $accountId);
                
                $db->commit();
                sc_inv_send(true, ['id' => $payId, 'payment_no' => $payNo], 'Pembayaran berhasil disimpan & dijurnal.');
            } catch (Throwable $e) {
                if ($db->inTransaction()) $db->rollBack();
                throw $e;
            }
            break;

        /* ------------------------------------------------------------------
         * STEP 2: CANCEL PAYMENT
         * ------------------------------------------------------------------ */
        case 'cancel_payment':
            $id = (int)($_POST['id'] ?? 0);
            $reason = trim((string)($_POST['reason'] ?? ''));
            if (strlen($reason) < 5) throw new ScInvError('Alasan pembatalan minimal 5 karakter.');
            
            $db->beginTransaction();
            try {
                $stmt = $db->prepare("SELECT * FROM subcon_invoice_payments WHERE id = :id FOR UPDATE");
                $stmt->execute([':id' => $id]);
                $pay = $stmt->fetch(PDO::FETCH_ASSOC);
                if (!$pay) throw new ScInvError('Pembayaran tidak ditemukan.');
                if ($pay['status'] !== 'Posted') throw new ScInvError('Pembayaran sudah dibatalkan.');
                
                if (sc_inv_has_journal($db, SC_INV_PAY_REF, $id)) {
                    ReverseJournalByReference($db, SC_INV_PAY_REF, $id);
                }
                
                $db->prepare("UPDATE subcon_invoice_payments SET status = 'Dibatalkan', cancelled_by = :uid,
                    cancelled_at = NOW(), cancel_reason = :reason WHERE id = :id")
                    ->execute([':uid' => sc_inv_uid(), ':reason' => $reason, ':id' => $id]);
                
                $stmt = $db->prepare("SELECT COALESCE(SUM(amount), 0) FROM subcon_invoice_payments 
                    WHERE invoice_id = :iid AND status = 'Posted'");
                $stmt->execute([':iid' => $pay['invoice_id']]);
                $newPaid = (float)$stmt->fetchColumn();
                $db->prepare("UPDATE subcon_invoices SET paid_amount = :paid WHERE id = :id")
                    ->execute([':paid' => $newPaid, ':id' => $pay['invoice_id']]);
                
                $db->commit();
                sc_inv_send(true, [], 'Pembayaran dibatalkan.');
            } catch (Throwable $e) {
                if ($db->inTransaction()) $db->rollBack();
                throw $e;
            }
            break;

        default:
            sc_inv_send(false, [], 'Task tidak dikenali.');
            break;

            var fAccount = new myComboField({ fieldLabel: 'Akun Kas/Bank', store: accountDs, displayField: 'label', valueField: 'id', mode: 'local', triggerAction: 'all', forceSelection: true, allowBlank: false, anchor: '95%' });
            var fMethod = new myComboField({ fieldLabel: 'Metode', store: new Ext.data.ArrayStore({ fields: ['v', 't'], data: [['BANK_TRANSFER', 'Transfer Bank'], ['CASH', 'Tunai'], ['CHEQUE', 'Cek']] }), displayField: 't', valueField: 'v', mode: 'local', triggerAction: 'all', forceSelection: true, value: 'BANK_TRANSFER', anchor: '95%' });
            var fAmount = new myNumberField({ fieldLabel: 'Jumlah Bayar', decimalPrecision: 2, allowNegative: false, value: inv.unpaid_amount, width: 150 });
            var fRef = new myTextField({ fieldLabel: 'No. Referensi', anchor: '95%' });
            var fNotes = new myTextField({ fieldLabel: 'Catatan', anchor: '95%' });
            
            var win = new Ext.Window({
                title: 'Pembayaran Invoice ' + inv.invoice_no, width: 500, height: 380, modal: true, layout: 'fit',
                items: new Ext.form.FormPanel({
                    border: false, labelWidth: 130, bodyStyle: 'padding:12px;',
                    items: [{ xtype: 'displayfield', hideLabel: true, value: 'Sisa hutang: ' + fmtAmt(inv.unpaid_amount) + '' },
                        fPayDate, fAccount, fMethod, fAmount, fRef, fNotes]
                }),
                bbar: ['->', { xtype: 'tbtext', text: '' +
                    '' }]
            });
            
            accountDs.load({ callback: function () { win.show(); } });
            
            setTimeout(function () {
                document.querySelector('.scPaySave').onclick = function () {
                    if (!fPayDate.getValue() || !fAccount.getValue()) { showErr('Tanggal dan akun wajib diisi.'); return; }
                    if (!(parseFloat(fAmount.getValue()) > 0)) { showErr('Jumlah harus > 0.'); return; }
                    
                    ajax({
                        task: 'save_payment', invoice_id: invoiceId, contract_id: inv.contract_id,
                        payment_date: ymd(fPayDate.getValue()), account_id: fAccount.getValue(),
                        payment_method: fMethod.getValue(), amount: fAmount.getValue(),
                        reference_no: fRef.getValue(), notes: fNotes.getValue()
                    }, function (res) {
                        if (res.success) {
                            win.close();
                            Ext.Msg.alert('Sukses', res.message);
                            invoiceDs.load();
                            loadDetailPanel(invoiceId);
                        } else showErr(res.message || 'Gagal.');
                    });
                };
                document.querySelector('.scPayCancel').onclick = function () { win.close(); };
            }, 100);
        });
    }

                Ext.Msg.alert(res.success ? 'Sukses' : 'Gagal', res.message || '');
